function run () {
    var pvStr = 'jceRRxJ';
    var endStr = 'jdc0XrO';
    var data = [];
    var More = configValue ? configValue : 1;

    var taobaoScheme = {
        scheme: "taobao",
        main: "taobao://m.taobao.com/tbopen/index.html?h5Url=https://m.taobao.com",
        start: "taobao://m.tb.cn/h." + pvStr,
        end: "taobao://m.tb.cn/h." + endStr
    };

    var ttScheme = {
        scheme: "taobaolite://m.ltao.com/open/index.html?action=ali.open.nav&h5Url=",
        main: "taobaolite://m.ltao.com/open/index.html?action=ali.open.nav&h5Url=https%3A%2F%2Fm.ltao.com%2Fhomepage%23mallPage",
        start: "taobao://m.taobao.com/tbopen/index.html?h5Url=https://m.tb.cn/h." + pvStr,
        end: "taobao://m.taobao.com/tbopen/index.html?h5Url=https://m.tb.cn/h." + endStr,
        type: "url"
    };

    var schemes = {1: [taobaoScheme], 2: [ttScheme]};
    var scheme = schemes[More] || schemes[1];
    var taskList = DATA.activity.taskList.map((v) => DATA.taskData[v]);

    scheme.forEach((s) => {
        taskList.forEach((oneTask) => {
            var items = oneTask.task.item;
            var before = oneTask.task.before || [];
            var baseUrl = oneTask.task.baseurl;
            var wait = oneTask.task.wait;
            var paramObj = oneTask.task.param || {};
            var mainIndex = oneTask.task.mainIndex;

            if (s.start) data.push({url: s.start, wait: 10});
            before.forEach((v) => {
                data.push({
                    url: s.type ? s.scheme + encodeURIComponent('https' + v.url) : s.scheme + v.url,
                    wait: v.wait
                });
            });
            data.push({url: s.main, wait: 5});

            items.forEach((item, i) => {
                var params = item.split(' ');
                var url = baseUrl;
                for (var key in paramObj) {
                    var values = paramObj[key];
                    var replacement = values[Math.floor(Math.random() * values.length)];
                    url = url.replace(new RegExp(key, 'g'), replacement);
                }
                params.forEach((value, index) => {
                    if (value) url = url.replace(new RegExp('占位参数' + (index + 1), 'g'), value);
                });
                data.push({
                    url: s.type ? s.scheme + encodeURIComponent('https' + url) : s.scheme + url,
                    wait: mainIndex && (i + 1) % mainIndex === 0 ? 23 : wait
                });
                if (mainIndex ? (i + 1) % mainIndex === 0 : i % 2 === 0) {
                    data.push({url: s.main, wait: 5});
                }
            });
            if (s.end) {
                if (mainIndex) data.push({url: s.main, wait: 5});
                data.push({url: s.end, wait: 5});
            }
        });
    });

    if (runNum > 0 && runNum != 1) data = data.slice(runNum - 1);
    return JSON.stringify({succes: 'ok', data: data});
};
